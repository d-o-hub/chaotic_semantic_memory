    #[test]
    #[cfg(feature = "ann-hnsw")]
    fn test_hnsw_index_drop_order_sequenced() -> Result<()> {
        use std::sync::Mutex;
        let drops = Arc::new(Mutex::new(Vec::new()));

        struct SequenceTracker(String, Arc<Mutex<Vec<String>>>);
        impl Drop for SequenceTracker {
            fn drop(&mut self) {
                self.1.lock().unwrap().push(self.0.clone());
            }
        }

        {
            let mut index = HnswIndex::new(16, 200, 50)?;

            // We can't easily track Hnsw's drop directly without changing its type,
            // but we can add another field to the owner to track when the owner is dropped.
            // Since hnsw is declared before _owner, it MUST be dropped before it.
            index._owner = Some(Box::new(SequenceTracker("owner".to_string(), drops.clone())));

            // To verify the field drop order, we can look at the struct definition.
            // But we can also add a temporary "first" tracker if we were to modify the struct.
            // Given the current struct:
            // hnsw: Hnsw,
            // ...
            // _owner: Option<Box<...>>
            // The drop order is guaranteed by Rust to be hnsw then _owner.
        }

        let order = drops.lock().unwrap();
        assert_eq!(order.len(), 1);
        assert_eq!(order[0], "owner");

        Ok(())
    }
